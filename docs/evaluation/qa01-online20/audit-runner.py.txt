def write(name, value):
    target = OUT / name
    tmp = target.with_suffix(target.suffix + '.tmp')
    tmp.write_text(json.dumps(value, ensure_ascii=False, indent=2), encoding='utf-8')
    tmp.replace(target)

def assets():
    paths = list((ROOT / 'outputs/m10-09-deepseek-live/runtime-v1').rglob('*'))
    paths += list((ROOT / 'outputs/m9-01/index-bge-m3-5617a9f6').rglob('*'))

    return {str(p.relative_to(ROOT)): hashlib.sha256(p.read_bytes()).hexdigest() for p in paths if p.is_file()}

def check_memory():
    class Status(ctypes.Structure):
        _fields_ = [('length',ctypes.c_ulong),('load',ctypes.c_ulong)] + [(n,ctypes.c_ulonglong) for n in ['total','avail','page_total','page_avail','virtual_total','virtual_avail','extended']]
    mem = Status(); mem.length = ctypes.sizeof(mem)
    if not ctypes.windll.kernel32.GlobalMemoryStatusEx(ctypes.byref(mem)) or mem.avail < 4*1024**3:
        raise RuntimeError('available memory below 4 GiB or unavailable')
    class Counters(ctypes.Structure):
        _fields_ = [('cb',ctypes.c_ulong),('faults',ctypes.c_ulong)] + [(n,ctypes.c_size_t) for n in ['peak','working','paged_peak','paged','nonpaged_peak','nonpaged','pagefile','pagefile_peak']]
    counters = Counters(); counters.cb = ctypes.sizeof(counters)
    ctypes.windll.kernel32.GetCurrentProcess.restype = ctypes.c_void_p
    ctypes.windll.psapi.GetProcessMemoryInfo.argtypes = [ctypes.c_void_p,ctypes.c_void_p,ctypes.c_ulong]
    if not ctypes.windll.psapi.GetProcessMemoryInfo(ctypes.windll.kernel32.GetCurrentProcess(),ctypes.byref(counters),counters.cb):
        raise RuntimeError('cannot inspect RSS')
    rss = counters.working
    if rss > 12*1024**3:
        raise RuntimeError('audit RSS above 12 GiB')
    return {'available_gib':round(mem.avail/1024**3,2),'rss_gib':round(rss/1024**3,2)}

class Ledger(_LiveCallLedger):
    current = None
    def save(self):
        write('call-ledger.json', {'head':HEAD,'max_calls':50,'calls':self.calls,'observations':self.observations})
    def begin(self, phase):
        check_memory()
        index = super().begin(phase)
        self.observations[index].update(case=self.current, started_at=datetime.now(timezone.utc).isoformat())
        self.save() # durable reservation BEFORE sending network request
        return index
    def response_received(self,index,response,latency_ms):
        super().response_received(index,response,latency_ms)
        usage = getattr(response,'usage',None)
        self.observations[index]['usage'] = usage.model_dump(mode='json') if hasattr(usage,'model_dump') else None
        self.save()
    def transport_failed(self,index,latency_ms):
        super().transport_failed(index,latency_ms); self.save()
